export async function initializeSession(): Promise<void> {
  if (!browser || initialized) {
    return;
  }
  initialized = true;
  const params = new URLSearchParams(window.location.search);
  scenario = params.get('scenario') ?? 'populated';
  const role = params.get('role') ?? 'admin';
  session.currentPrincipal = role === 'logged-out' ? undefined : role === 'member' ? member : admin;
  const workspace = params.get('workspace') ?? 'prod';
  session.workspaces = workspace === 'none' ? undefined : workspaceDirectory({ ...config, current_workspace: workspace });
  session.isInitializing = false;
  if (scenario === 'empty') {
    targets = [];
    grants = [];
  }
}
export async function waitForSession(): Promise<void> { await initializeSession(); }
export async function signIn(): Promise<void> { throw new Error('SSO desabilitado na prévia visual.'); }
export async function signOut(): Promise<void> { session.currentPrincipal = undefined; }
export function clearSessionError(): void { session.sessionError = undefined; }
export async function apiFetch(path: string, init: RequestInit = {}): Promise<Response> {
  await initializeSession();
  await new Promise((resolve) => setTimeout(resolve, 120));
  if (scenario === 'error') {
    throw new Error('Falha simulada: serviço temporariamente indisponível.');
  }
  const method = init.method ?? 'GET';
  if (method !== 'GET') {
    // Do not simulate successful writes: a fixture is not persistence verification.
    throw new Error('Prévia visual: alterações não são enviadas a um servidor real.');
  }
  if (path === '/v1/auth/config') { return Response.json(config); }
  if (path.startsWith('/v1/metrics?')) {
    const requested = new URL(path, 'http://127.0.0.1').searchParams.get('period');
    const period = requested === '7d' ? '7d' : requested === '30d' ? '30d' : '24h';
    const data = metricsFixture(period, session.currentPrincipal?.platform_role === 'admin' ? 'workspace' : 'personal', scenario === 'empty');
    return Response.json(scenario === 'malformed' ? { ...data, summary: { total_queries: -1 } } : data);
  }
  if (path === '/v1/me') { return Response.json({ principal: session.currentPrincipal, database_grants: grants }); }
  if (path === '/v1/admin/database-targets') { return Response.json(targets); }
  if (path === '/v1/admin/principals') { return Response.json([admin, member]); }
  if (path.endsWith('/database-grants')) { return Response.json(grants); }
  if (path.startsWith('/v1/admin/audit-events')) { return Response.json(scenario === 'empty' ? [] : events); }
  throw new Error(`Rota não implementada na prévia: ${path}`);
}
